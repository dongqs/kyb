# TTS Server — __PORT__
# macOS only. Requires `say` and `afplay`.
# Port: TTS_PORT env var (default 10666)

# Endpoints
GET  /health   Health check
GET  /voices   List voices
GET  /ping     Play alert
GET  /speak    Speak text (GET query or POST JSON)
GET  /done     Speak + alert

# Full OpenAPI spec
/openapi.yml

# English example (use a English voice)
curl "http://localhost:__PORT__/speak?text=Hello+world&voice=Zarvox&rate=200"

# Chinese example (use a Chinese voice)
curl "http://localhost:__PORT__/speak?text=%E4%BD%A0%E5%A5%BD%E4%B8%96%E7%95%8C&voice=Tingting&rate=200"
# or via POST
curl -X POST http://localhost:__PORT__/speak \
  -H "Content-Type: application/json" \
  -d '{"text":"你好世界","voice":"Tingting","rate":200}'

# Done notification
curl "http://localhost:__PORT__/done?text=Build+succeeded"

# Each voice only supports specific languages.
# Run GET /voices to see the language column and pick a matching voice.
